   144→    function animateCounter(el: HTMLElement) {
   145→      if (el.dataset.animated) return;
   146→      el.dataset.animated = 'true';
   147→      const target = parseFloat(el.dataset.target || '0');
   148→      const isDecimal = el.dataset.decimal === '1';
   149→      const duration = 2000;
   150→      const start = performance.now();
   151→      function step(now: number) {
   152→        const elapsed = now - start;
   153→        const progress = Math.min(elapsed / duration, 1);
   154→        const eased = 1 - Math.pow(1 - progress, 3);
   155→        const value = target * eased;
   156→        if (isDecimal) el.textContent = (value / 10).toFixed(1);
   157→        else el.textContent = String(Math.floor(value));
   158→        if (progress < 1) requestAnimationFrame(step);
   159→        else el.textContent = isDecimal ? (target / 10).toFixed(1) : String(target);
   160→      }
   161→      requestAnimationFrame(step);
   162→    }

   164→    // QR Code generation
   165→    function generateQRPattern(size = 21) {
   166→      const pattern: number[][] = [];
   167→      for (let i = 0; i < size; i++) pattern[i] = new Array(size).fill(0);
   168→      const addFinder = (sr: number, sc: number) => {
   169→        for (let i = 0; i < 7; i++) for (let j = 0; j < 7; j++) {
   170→          if (i === 0 || i === 6 || j === 0 || j === 6 || (i >= 2 && i <= 4 && j >= 2 && j <= 4)) pattern[sr + i][sc + j] = 1;
   171→        }
   172→      };
   173→      addFinder(0, 0); addFinder(0, size - 7); addFinder(size - 7, 0);
   174→      for (let i = 8; i < size - 8; i++) { pattern[6][i] = i % 2 === 0 ? 1 : 0; pattern[i][6] = i % 2 === 0 ? 1 : 0; }
   175→      let seed = 12345;
   176→      const rand = () => { seed = (seed * 9301 + 49297) % 233280; return seed / 233280; };
   177→      for (let i = 0; i < size; i++) for (let j = 0; j < size; j++) {
   178→        if ((i < 8 && j < 8) || (i < 8 && j >= size - 8) || (i >= size - 8 && j < 8)) continue;
   179→        if (i === 6 || j === 6) continue;
   180→        if (rand() > 0.5) pattern[i][j] = 1;
   181→      }
   182→      return pattern;
   183→    }

   185→    function renderQR(container: HTMLElement | null, color: string, template: string) {
   186→      if (!container) return;
   187→      const pattern = generateQRPattern();
   188→      const size = pattern.length;
   189→      container.innerHTML = '';
   190→      container.style.display = 'grid';
   191→      container.style.gridTemplateColumns = `repeat(${size}, 1fr)`;
   192→      container.style.gridTemplateRows = `repeat(${size}, 1fr)`;
   193→      container.style.gap = '0';
   194→      let borderRadius = '0';
   195→      if (template === 'rounded') borderRadius = '30%';
   196→      if (template === 'dot') borderRadius = '50%';
   197→      const frag = document.createDocumentFragment();
   198→      for (let i = 0; i < size; i++) for (let j = 0; j < size; j++) {
   199→        const cell = document.createElement('div');
   200→        cell.className = 'qr-cell';
   201→        if (pattern[i][j] === 1) { cell.style.background = color; cell.style.borderRadius = borderRadius; }
   202→        frag.appendChild(cell);
   203→      }
   204→      container.appendChild(frag);
   205→    }
