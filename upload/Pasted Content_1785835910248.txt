<script>
  lucide.createIcons();

  // Nomor WhatsApp Pemilik Toko (Ganti dengan nomor sungguhan)
  const WARUNG_WA = '6281234567890';

  // Sticky shadow
  const stickyWrap = document.getElementById('sticky-wrap');
  window.addEventListener('scroll', () => {
    if (window.scrollY > 20) stickyWrap.classList.add('scrolled');
    else stickyWrap.classList.remove('scrolled');
  });

  // Fade-up entry
  const fadeEls = document.querySelectorAll('.fade-up');
  const io = new IntersectionObserver((entries) => {
    entries.forEach((e, i) => {
      if (e.isIntersecting) {
        setTimeout(() => e.target.classList.add('in'), i * 60);
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.1, rootMargin: '0px 0px -20px 0px' });
  fadeEls.forEach(el => io.observe(el));

  // Category & Search Filter
  const chips = document.querySelectorAll('.cat-chip');
  const menuCards = document.querySelectorAll('.menu-card');
  const emptyState = document.getElementById('empty-state');
  const menuCount = document.getElementById('menu-count');
  let currentCat = 'all';
  let currentSearch = '';

  function applyFilter() {
    let visible = 0;
    menuCards.forEach(card => {
      const cat = card.dataset.cat;
      const name = card.dataset.name || '';
      const catMatch = currentCat === 'all' || cat === currentCat;
      const searchMatch = currentSearch === '' || name.includes(currentSearch.toLowerCase());
      if (catMatch && searchMatch) {
        card.style.display = '';
        visible++;
      } else {
        card.style.display = 'none';
      }
    });
    emptyState.classList.toggle('hidden', visible > 0);
    menuCount.textContent = `${visible} menu tersedia`;
  }

  chips.forEach(chip => {
    chip.addEventListener('click', () => {
      chips.forEach(c => c.classList.remove('active'));
      chip.classList.add('active');
      currentCat = chip.dataset.cat;
      applyFilter();
    });
  });

  const searchInput = document.getElementById('search-input');
  const searchClear = document.getElementById('search-clear');
  searchInput.addEventListener('input', (e) => {
    currentSearch = e.target.value.trim();
    searchClear.classList.toggle('hidden', currentSearch === '');
    searchClear.classList.toggle('flex', currentSearch !== '');
    applyFilter();
  });
  searchClear.addEventListener('click', () => {
    searchInput.value = '';
    currentSearch = '';
    searchClear.classList.add('hidden');
    searchClear.classList.remove('flex');
    applyFilter();
  });

  // ====== CART SYSTEM ======
  let cart = {};
  const cartBar = document.getElementById('cart-bar');

  function addToCart(name, price) {
    if (cart[name]) {
      cart[name].qty += 1;
    } else {
      cart[name] = { name, price, qty: 1 };
    }
    updateCart();
    updateMenuButtons();
    updateDetailButtonIfOpen(name, price);
    showToast(`${name} ditambahkan`);
  }

  function removeFromCart(name) {
    if (cart[name]) {
      cart[name].qty -= 1;
      if (cart[name].qty <= 0) {
        delete cart[name];
      }
      updateCart();
      updateMenuButtons();
      updateDetailButtonIfOpen(name, 0);
    }
  }

  function getCartCount() {
    return Object.values(cart).reduce((sum, item) => sum + item.qty, 0);
  }

  function getCartTotal() {
    return Object.values(cart).reduce((sum, item) => sum + (item.price * item.qty), 0);
  }

  function updateCart() {
    const count = getCartCount();
    const total = getCartTotal();
    if (count === 0) {
      cartBar.classList.remove('show');
    } else {
      cartBar.classList.add('show');
      document.getElementById('cart-count').textContent = count;
      document.getElementById('cart-items').textContent = `${count} item`;
      document.getElementById('cart-total').textContent = `Rp ${total.toLocaleString('id-ID')}`;
    }
  }

  function updateMenuButtons() {
    document.querySelectorAll('.menu-card').forEach(card => {
      const name = card.dataset.fullname;
      const price = parseInt(card.dataset.price);
      const isHabis = card.dataset.habis === 'true';
      const btnWrap = card.querySelector('.btn-wrap');
      
      if (!btnWrap) return;
      
      if (isHabis) {
        btnWrap.innerHTML = `<span class="text-[11px] font-bold text-slate-400 italic">Habis</span>`;
        return;
      }
      
      if (cart[name]) {
        btnWrap.innerHTML = `
          <div class="qty-stepper">
            <button onclick="removeFromCart('${name.replace(/'/g, "\\'")}')"><i data-lucide="minus" class="w-4 h-4" stroke-width="2.5"></i></button>
            <span>${cart[name].qty}</span>
            <button onclick="addToCart('${name.replace(/'/g, "\\'")}', ${price})"><i data-lucide="plus" class="w-4 h-4" stroke-width="2.5"></i></button>
          </div>
        `;
      } else {
        btnWrap.innerHTML = `
          <button onclick="addToCart('${name.replace(/'/g, "\\'")}', ${price})" class="add-btn"><i data-lucide="plus" class="w-5 h-5" stroke-width="2.5"></i></button>
        `;
      }
    });
    lucide.createIcons();
  }

  // ====== LIGHTBOX ======
  function openLightbox(imgSrc) {
    document.getElementById('lightbox-img').src = imgSrc;
    document.getElementById('lightbox').classList.add('show');
    document.body.style.overflow = 'hidden';
  }

  function closeLightbox() {
    document.getElementById('lightbox').classList.remove('show');
    document.body.style.overflow = '';
  }

  // ====== DETAIL MENU MODAL ======
  function openDetail(el) {
    const card = el.closest('.menu-card');
    if (!card || card.dataset.habis === 'true') return;
    
    const fullname = card.dataset.fullname;
    const price = parseInt(card.dataset.price);
    const desc = card.dataset.desc;
    const img = card.dataset.img;
    
    document.getElementById('detail-img').src = img;
    document.getElementById('detail-name').textContent = fullname;
    document.getElementById('detail-price').textContent = `Rp ${price.toLocaleString('id-ID')}`;
    document.getElementById('detail-desc').textContent = desc;
    
    updateDetailButton(fullname, price);
    
    document.getElementById('detail-backdrop').classList.add('show');
    document.getElementById('detail-sheet').classList.add('show');
  }

  function closeDetail() {
    document.getElementById('detail-backdrop').classList.remove('show');
    document.getElementById('detail-sheet').classList.remove('show');
  }
  document.getElementById('detail-backdrop').addEventListener('click', closeDetail);

  function updateDetailButton(name, price) {
    const btnWrap = document.getElementById('detail-btn-wrap');
    if (cart[name]) {
      btnWrap.innerHTML = `
        <div class="qty-stepper wide w-full justify-between">
          <button onclick="removeFromCart('${name.replace(/'/g, "\\'")}')"><i data-lucide="minus" class="w-5 h-5" stroke-width="2.5"></i></button>
          <span>${cart[name].qty}</span>
          <button onclick="addToCart('${name.replace(/'/g, "\\'")}', ${price})"><i data-lucide="plus" class="w-5 h-5" stroke-width="2.5"></i></button>
        </div>
      `;
    } else {
      btnWrap.innerHTML = `
        <button onclick="addToCart('${name.replace(/'/g, "\\'")}', ${price})" class="w-full py-3 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 text-white font-bold text-sm shadow-lg shadow-amber-500/30 flex items-center justify-center gap-2 active:scale-95 transition-transform">
          <i data-lucide="plus" class="w-5 h-5" stroke-width="2.5"></i>
          Tambah ke Keranjang
        </button>
      `;
    }
    lucide.createIcons();
  }

  function updateDetailButtonIfOpen(name, price) {
    if (document.getElementById('detail-sheet').classList.contains('show')) {
      const currentName = document.getElementById('detail-name').textContent;
      if (currentName === name) {
        const itemPrice = cart[name] ? cart[name].price : price;
        updateDetailButton(name, itemPrice);
      }
    }
  }

  // ====== CHECKOUT MODAL ======
  let orderType = 'dinein';

  function openCheckout() {
    if (Object.keys(cart).length === 0) {
      showToast('Keranjang masih kosong, yuk pilih menu dulu!');
      return;
    }

    const summaryItems = document.getElementById('summary-items');
    summaryItems.innerHTML = '';
    for (const key in cart) {
      const item = cart[key];
      const subTotal = item.price * item.qty;
      summaryItems.innerHTML += `
        <div class="flex items-center justify-between text-xs">
          <span class="text-slate-600 truncate flex-1 pr-2">${item.name} <span class="text-slate-400">x${item.qty}</span></span>
          <span class="font-semibold text-slate-900">Rp ${subTotal.toLocaleString('id-ID')}</span>
        </div>
      `;
    }
    document.getElementById('summary-total').textContent = `Rp ${getCartTotal().toLocaleString('id-ID')}`;
    
    document.getElementById('checkout-backdrop').classList.add('show');
    document.getElementById('checkout-sheet').classList.add('show');
  }

  function closeCheckout() {
    document.getElementById('checkout-backdrop').classList.remove('show');
    document.getElementById('checkout-sheet').classList.remove('show');
  }
  document.getElementById('checkout-backdrop').addEventListener('click', closeCheckout);

  function selectType(type) {
    orderType = type;
    const dineinBtn = document.getElementById('opt-dinein');
    const takeawayBtn = document.getElementById('opt-takeaway');
    const tableWrap = document.getElementById('table-input-wrap');
    
    if (type === 'dinein') {
      dineinBtn.classList.add('active');
      takeawayBtn.classList.remove('active');
      tableWrap.classList.remove('is-hidden');
    } else {
      takeawayBtn.classList.add('active');
      dineinBtn.classList.remove('active');
      tableWrap.classList.add('is-hidden');
    }
  }

  // Send to WhatsApp
  function sendToWhatsApp() {
    const name = document.getElementById('cust-name').value.trim() || 'Tanpa Nama';
    const table = document.getElementById('cust-table').value.trim();
    const notes = document.getElementById('cust-notes').value.trim();
    
    if (orderType === 'dinein' && !table) {
      showToast('Mohon isi nomor meja dulu');
      return;
    }

    let message = `*PESANAN BARU - AYAM GEPREK AAN MUDA*\n\n`;
    message += `*Nama Pemesan:* ${name}\n`;
    message += `*Tipe:* ${orderType === 'dinein' ? 'Makan di Tempat' : 'Bawa Pulang'}\n`;
    if (orderType === 'dinein') message += `*Nomor Meja:* ${table}\n`;
    if (notes) message += `*Catatan:* ${notes}\n`;
    message += `\n*Detail Pesanan:*\n`;
    
    let i = 1;
    for (const key in cart) {
      const item = cart[key];
      const subTotal = item.price * item.qty;
      message += `${i}. ${item.name} x${item.qty} - Rp ${subTotal.toLocaleString('id-ID')}\n`;
      i++;
    }
    
    const total = getCartTotal();
    message += `\n*Total Pembayaran: Rp ${total.toLocaleString('id-ID')}*\n\n`;
    message += `_Pesanan dibuat via Menu Digital PesanLagi_`;

    const waUrl = `https://wa.me/${WARUNG_WA}?text=${encodeURIComponent(message)}`;
    window.open(waUrl, '_blank');
    
    cart = {};
    updateCart();
    updateMenuButtons();
    closeCheckout();
    showToast('Pesanan dikirim ke WhatsApp warung!');
  }

  // Toast
  let toastTimer;
  function showToast(msg) {
    const toast = document.getElementById('toast');
    document.getElementById('toast-text').textContent = msg;
    toast.style.opacity = '1';
    toast.style.transform = 'translateX(-50%) translateY(0)';
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => {
      toast.style.opacity = '0';
      toast.style.transform = 'translateX(-50%) translateY(10px)';
    }, 1800);
  }

  // Init buttons on load
  updateMenuButtons();
</script>
