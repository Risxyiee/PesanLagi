function sendReply(){
  const c=CONVS.find(x=>x.id===activeConvId); if(!c) return;
  const body=$('cInput').value.trim(); if(!body) return;
  const now=new Date().toISOString();
  c.messages.push({sender:'admin',body,created_at:now});
  c.status='manual'; c.last_message_at=now; c.unread=0;
  save(); renderInbox();
  setTimeout(showTyping,1100);
  setTimeout(()=>{
    removeTyping();
    const cc=CONVS.find(x=>x.id===activeConvId); if(!cc) return;
    cc.messages.push({sender:'customer',body:CUST_REPLIES[hash(cc.id+cc.messages.length)%CUST_REPLIES.length],created_at:new Date().toISOString()});
    cc.last_message_at=new Date().toISOString(); save(); renderInbox();
  },2700);
}

  r.readAsDataURL(file);
}

function detectDelim(text){
  const line=(text.split(/\r?\n/).find(l=>l.trim())||'');
  const s=(line.match(/;/g)||[]).length,c=(line.match(/,/g)||[]).length,t=(line.match(/\t/g)||[]).length;
  if(t>s&&t>c) return '\t'; return s>=c?';':',';
}
function parseCSVText(text){
  const d=detectDelim(text); const rows=[]; let row=[],cell='',q=false;
  for(let i=0;i<text.length;i++){ const ch=text[i];
    if(q){ if(ch==='"'){ if(text[i+1]==='"'){cell+='"';i++;} else q=false; } else cell+=ch; }
    else { if(ch==='"')q=true; else if(ch===d){row.push(cell);cell='';} else if(ch==='\n'){row.push(cell);rows.push(row);row=[];cell='';} else if(ch!=='\r')cell+=ch; } }
  if(cell!==''||row.length){row.push(cell);rows.push(row);}
  return rows.filter(r=>r.some(c=>String(c).trim()!==''));
}
function mapStock(v){
  const s=String(v||'').trim().toLowerCase();
  if(!s) return 10;
  if(/habis|kosong|out|sold/.test(s)) return 0;
  if(/menipis|sedikit|low/.test(s)) return 3;
  if(/tersedia|ready|ada|available|in/.test(s)) return 10;
  const d=s.replace(/[^\d]/g,'');
  if(/^\d+$/.test(d)) return Number(d);
  return null;
}

function qrSvg(seed){
  const N=21,C=7; let s=seed;
  const rnd=()=>{s=(s*1103515245+12345)%2147483648;return s/2147483648;};
  const fnd=(x,y)=>(x<8&&y<8)||(x>=N-8&&y<8)||(x<8&&y>=N-8);
  let r='';
  for(let y=0;y<N;y++)for(let x=0;x<N;x++){ if(fnd(x,y))continue; if(rnd()<.42)r+='<rect x="'+x*C+'" y="'+y*C+'" width="'+C+'" height="'+C+'" fill="#1C1917"/>'; }
  const eye=(fx,fy)=>{r+='<rect x="'+fx*C+'" y="'+fy*C+'" width="'+7*C+'" height="'+7*C+'" fill="#1C1917"/><rect x="'+(fx+1)*C+'" y="'+(fy+1)*C+'" width="'+5*C+'" height="'+5*C+'" fill="#fff"/><rect x="'+(fx+2)*C+'" y="'+(fy+2)*C+'" width="'+3*C+'" height="'+3*C+'" fill="#1C1917"/>';};
  eye(0,0); eye(N-7,0); eye(0,N-7);
  const CORN='position:absolute;width:22px;height:22px;border:3px solid #F97316';
  return '<div style="position:relative;padding:14px">'+
    '<span style="'+CORN+';top:0;left:0;border-right:0;border-bottom:0;border-radius:8px 0 0 0"></span>'+
    '<span style="'+CORN+';top:0;right:0;border-left:0;border-bottom:0;border-radius:0 8px 0 0"></span>'+
    '<span style="'+CORN+';bottom:0;left:0;border-right:0;border-top:0;border-radius:0 0 0 8px"></span>'+
    '<span style="'+CORN+';bottom:0;right:0;border-left:0;border-top:0;border-radius:0 0 8px 0"></span>'+
    '<svg viewBox="0 0 147 147" style="height:170px;width:170px">'+r+'</svg></div>';
}
