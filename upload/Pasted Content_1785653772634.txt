      }

      grid.innerHTML = menus.map(m => `
        <div class="bg-white rounded-2xl border border-slate-100 shadow-sm overflow-hidden group cursor-move" draggable="true" ondragstart="dragStart(event, '${m.id}')" ondragover="dragOver(event)" ondrop="drop(event, '${m.id}')">

            <img src="${m.image_url}" class="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300" alt="${m.name}">

              <svg class="w-4 h-4 text-slate-600" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 9l-3 3m0 0l3 3m-3-3h18m0 0l-3-3m3 3l-3 3"/></svg>

          <div class="p-4">
            <div class="flex items-start justify-between gap-2 mb-1">

    // Drag & Drop Handlers
    function dragStart(e, id) {
      draggedId = id;
      e.target.classList.add('dragging');
    }

    function dragOver(e) {
      e.preventDefault();
      e.currentTarget.classList.add('drag-over');
    }

    function drop(e, targetId) {
      e.preventDefault();
      e.currentTarget.classList.remove('drag-over');
      
      if (draggedId && draggedId !== targetId) {
        const draggedIndex = InsForgeDB.menus.findIndex(m => m.id === draggedId);
        const targetIndex = InsForgeDB.menus.findIndex(m => m.id === targetId);
        
        // Remove dragged item and insert at target index
        const [draggedItem] = InsForgeDB.menus.splice(draggedIndex, 1);
        InsForgeDB.menus.splice(targetIndex, 0, draggedItem);
        
        renderMenus();
        showToast('Urutan menu berhasil diubah!');
      }
    }

    function openMenuModal() {
      document.getElementById('modal-title').innerText = 'Tambah Menu Baru';
      document.getElementById('menu-form').reset();
      document.getElementById('menu-id').value = '';
      document.getElementById('menu-img-preview').src = 'https://picsum.photos/seed/foodplaceholder/200/200.jpg';
      document.getElementById('menu-available').checked = true;
      renderCategories(); // Ensure dropdown is populated
      document.getElementById('menu-modal').classList.remove('hidden');
    }

    function editMenu(id) {
      const m = InsForgeDB.menus.find(x => x.id === id);
      if (!m) return;
      
      document.getElementById('modal-title').innerText = 'Edit Menu';
      document.getElementById('menu-id').value = m.id;
      document.getElementById('menu-name').value = m.name;
      document.getElementById('menu-desc').value = m.desc;
      document.getElementById('menu-price').value = m.price;
      document.getElementById('menu-cat').value = m.category;
      document.getElementById('menu-img-preview').src = m.image_url;
      document.getElementById('menu-available').checked = m.is_available;
      
      document.getElementById('menu-modal').classList.remove('hidden');
    }

    function deleteMenu(id) {
      menuToDelete = id;
      document.getElementById('delete-modal').classList.remove('hidden');
    }

    document.getElementById('confirm-delete-btn').addEventListener('click', () => {
      if (menuToDelete) {
        InsForgeDB.deleteMenu(menuToDelete);
        renderMenus();
        showToast('Menu berhasil dihapus.');
        menuToDelete = null;
      }
      closeModal('delete-modal');
    });

    function handleMenuImageUpload(e) {
      const file = e.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = (ev) => {
          document.getElementById('menu-img-preview').src = ev.target.result;
          showToast('Foto diupload ke InsForge Storage...');
        };
        reader.readAsDataURL(file);
      }
    }

    function saveMenu(e) {
      e.preventDefault();
      const id = document.getElementById('menu-id').value;
      const menuData = {
        id: id || null,
        name: document.getElementById('menu-name').value,
        desc: document.getElementById('menu-desc').value,
        price: parseInt(document.getElementById('menu-price').value),
        category: document.getElementById('menu-cat').value,
        image_url: document.getElementById('menu-img-preview').src,
        is_available: document.getElementById('menu-available').checked
      };
      
      InsForgeDB.saveMenu(menuData);
      closeMenuModal();
      renderMenus();
      showToast(id ? 'Menu berhasil diperbarui!' : 'Menu baru berhasil ditambahkan!');
    }

    function generateQRCode() {
      const qrContainer = document.getElementById('qrcode');
      if (!qrContainer) return;
      qrContainer.innerHTML = '';
      const slug = InsForgeDB.stores.slug;
      const url = `https://pesanlagi.web.id/menu/${slug}`;
      
      new QRCode(qrContainer, {
        text: url,
        width: 120,
        height: 120,
        colorDark: document.getElementById('qr-color').value,
        colorLight: "#ffffff",
        correctLevel: QRCode.CorrectLevel.H
      });
    }

    function applyPreset(presetName) {
      let bg, qr;
      if (presetName === 'kopi-susu') { bg = '#E8D0B3'; qr = '#4E342E'; isCustomColor = false; }
      else if (presetName === 'sage-segar') { bg = '#B2AC88'; qr = '#1A1A1A'; isCustomColor = false; }
      else if (presetName === 'midnight-orange') { bg = '#1A1A1A'; qr = '#FF6D00'; isCustomColor = false; }
      else if (presetName === 'neon-cyber') { bg = '#0F0F0F'; qr = '#00F0FF'; isCustomColor = true; }
      else if (presetName === 'warm-pastel') { bg = '#FFE4E6'; qr = '#DB2777'; isCustomColor = true; }
      else if (presetName === 'minimalist-black') { bg = '#FFFFFF'; qr = '#000000'; isCustomColor = true; }
      
      document.getElementById('bg-color').value = bg;
      document.getElementById('qr-color').value = qr;
      
      updateCardUI();
      checkFreemiumLogic();
    }

    function setBgColor(color) {
      isCustomColor = true;
      document.getElementById('bg-color').value = color;
      updateCardUI();
      checkFreemiumLogic();
    }

      isCustomColor = true;
      document.getElementById('qr-color').value = color;
      updateCardUI();
      checkFreemiumLogic();
    }

    function applyCustomColor() {
      isCustomColor = true;
      updateCardUI();
      checkFreemiumLogic();
    }

    function updateCardUI() {
      const bg = document.getElementById('bg-color').value;
      const qr = document.getElementById('qr-color').value;
      const tableNum = document.getElementById('table-number').value;
      
      const card = document.getElementById('qr-card');
      card.style.backgroundColor = bg;
      
      const texts = card.querySelectorAll('h4, p');
      texts.forEach(t => t.style.color = qr);
      
      const tableNumEl = document.getElementById('qr-table-number');
      if (tableNum.trim()) {
        tableNumEl.textContent = tableNum;
        tableNumEl.classList.remove('hidden');
        tableNumEl.style.color = qr;
        tableNumEl.style.backgroundColor = `rgba(255,255,255,0.2)`;
      } else {
        tableNumEl.classList.add('hidden');
      }
      
      generateQRCode();
    }

    function checkFreemiumLogic() {
      const isPro = InsForgeDB.users.is_pro;
      const watermark = document.getElementById('watermark');
      const proBadge = document.getElementById('pro-badge-custom');
      
      if (!isPro && isCustomColor) {
        watermark.classList.remove('hidden');
        proBadge.classList.remove('hidden');
      } else {
        watermark.classList.add('hidden');
        if (!isCustomColor) proBadge.classList.add('hidden');
      }
    }

    function handleSaveDesign() {
      const isPro = InsForgeDB.users.is_pro;
      const bg = document.getElementById('bg-color').value;
      const qr = document.getElementById('qr-color').value;
      
      if (!isPro && isCustomColor) {
        document.getElementById('upgrade-modal').classList.remove('hidden');
      } else {
        InsForgeDB.saveStore({ bg_color: bg, qr_color: qr });

      }
    }

    function setLoading(btn, isLoading, text) {
      if (!btn) return;
      if (isLoading) {
        btn.disabled = true;
        btn.dataset.originalText = btn.innerHTML;
        btn.innerHTML = `<svg class="animate-spin inline-block w-5 h-5 mr-2" fill="none" viewBox="0 0 24 24"><circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle><path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path></svg>${text}`;
      } else {
        btn.disabled = false;
        btn.innerHTML = btn.dataset.originalText;
      }
    }
