<script>
  lucide.createIcons();

  const pageTitles = { dashboard:'Dashboard Warung', menu:'Menu Makanan', qr:'Kartu QR Code', orders:'Pesanan Masuk', reports:'Laporan Penjualan', reviews:'Ulasan Pelanggan', settings:'Pengaturan Profil' };
  const sidebar = document.getElementById('sidebar');
  const backdrop = document.getElementById('backdrop');
  const main = document.getElementById('main-content');

  // Navigation
  function navigate(pageId) {
    document.querySelectorAll('.page-section').forEach(p => p.classList.remove('active'));
    const section = document.querySelector(`[data-page="${pageId}"]`);
    if (section) section.classList.add('active');
    document.querySelectorAll('.nav-item').forEach(i => i.classList.remove('active'));
    const navItem = document.querySelector(`.nav-item[data-nav="${pageId}"]`);
    if (navItem) navItem.classList.add('active');
    document.querySelectorAll('.bottom-nav-item').forEach(i => i.classList.remove('active'));
    const bottomItem = document.querySelector(`.bottom-nav-item[data-nav="${pageId}"]`);
    if (bottomItem) bottomItem.classList.add('active');
    const titleEl = document.getElementById('page-title');
    if (titleEl) titleEl.textContent = pageTitles[pageId] || 'Dashboard';
    // Retrigger stagger
    if (section) {
      section.querySelectorAll('.stagger').forEach(s => { s.classList.remove('stagger'); void s.offsetWidth; s.classList.add('stagger'); });
    }
    // Charts
    if (pageId === 'reports') { animateCharts(); }
    // Close mobile
    closeMobile();
    // Scroll top
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

  // Sidebar controls
  document.getElementById('hamburger').addEventListener('click', () => { sidebar.classList.add('mobile-open'); backdrop.classList.add('show'); });
  backdrop.addEventListener('click', closeMobile);
  function closeMobile() { sidebar.classList.remove('mobile-open'); backdrop.classList.remove('show'); }
  document.getElementById('collapse-btn').addEventListener('click', toggleCollapse);
  document.getElementById('collapse-btn-exp').addEventListener('click', toggleCollapse);
  function toggleCollapse() { sidebar.classList.toggle('collapsed'); main.classList.toggle('collapsed'); }

  // Nav click
  document.querySelectorAll('.nav-item').forEach(item => item.addEventListener('click', () => navigate(item.dataset.nav)));
  document.querySelectorAll('.bottom-nav-item').forEach(item => item.addEventListener('click', () => navigate(item.dataset.nav)));

  // Toggles
  function toggleStore(el) {
    el.classList.toggle('on');
    const isOn = el.classList.contains('on');
    document.getElementById('store-status').textContent = isOn ? 'Toko Buka' : 'Toko Tutup';
    document.getElementById('store-hours').textContent = isOn ? 'Buka sejak 08:00 WIB' : 'Tutup sementara';
    document.getElementById('store-hours').className = `text-[10px] font-semibold mt-1 ${isOn ? 'text-green-600' : 'text-red-500'}`;
  }
  function toggleStock(el, textId, onText, offText) {
    el.classList.toggle('on');
    const isOn = el.classList.contains('on');
    const text = document.getElementById(textId);
    text.textContent = isOn ? onText : offText;
    text.className = `hidden sm:inline text-xs font-semibold ${isOn ? 'text-green-600' : 'text-red-500'}`;
  }
  function toggleStockCard(el) { el.classList.toggle('on'); }

  // Order tabs
  const orderTabs = document.querySelectorAll('.order-tab');
  const orderIndicator = document.getElementById('order-tab-indicator');
  function updateOrderIndicator() {
    const active = document.querySelector('.order-tab.active');
    if (!active) return;
    orderIndicator.style.left = active.offsetLeft + 'px';
    orderIndicator.style.width = active.offsetWidth + 'px';

  // QR generation
  function generateQR() {
    const svg = document.getElementById('qr-preview-svg');
    if (!svg) return;
    const size = 25;
    let html = '';
    // Finder patterns
    [[0,0],[18,0],[0,18]].forEach(([fx,fy]) => {
      html += `<rect x="${fx}" y="${fy}" width="7" height="7" fill="#0F172A"/>`;
      html += `<rect x="${fx+1}" y="${fy+1}" width="5" height="5" fill="white"/>`;
      html += `<rect x="${fx+2}" y="${fy+2}" width="3" height="3" fill="#0F172A"/>`;
    });
    // Alignment pattern
    html += `<rect x="16" y="16" width="5" height="5" fill="#0F172A"/>`;
    html += `<rect x="17" y="17" width="3" height="3" fill="white"/>`;
    html += `<rect x="18" y="18" width="1" height="1" fill="#0F172A"/>`;
    // Timing patterns
    for (let i = 8; i < 17; i++) {
      if (i % 2 === 0) {
        html += `<rect x="${i}" y="6" width="1" height="1" fill="#0F172A"/>`;
        html += `<rect x="6" y="${i}" width="1" height="1" fill="#0F172A"/>`;
      }
    }
    // Data cells
    for (let y = 0; y < size; y++) {
      for (let x = 0; x < size; x++) {
        const inFinder = (x<8&&y<8)||(x>16&&y<8)||(x<8&&y>16);
        const inAlign = (x>=16&&x<=20&&y>=16&&y<=20);
        const inCenter = (x>=10&&x<=14&&y>=10&&y<=14);
        const inTiming = (y===6||x===6);
        if (inFinder||inAlign||inCenter||inTiming) continue;
        const hash = (x*7+y*13+x*y*3) % 7;
        if (hash < 3) html += `<rect x="${x}" y="${y}" width="1" height="1" fill="#0F172A"/>`;
      }
    }
    svg.innerHTML = html;
  }
  generateQR();

  // Charts animation
  function animateCharts() {
    const line = document.getElementById('chart-line');
    const area = document.getElementById('chart-area');
    const dots = document.querySelectorAll('.chart-dot');
    const bars = document.querySelectorAll('.bar-fill');
    // Reset
    if (line) { line.classList.remove('animate'); void line.offsetWidth; line.classList.add('animate'); }
    if (area) { area.style.opacity = '0'; area.style.transition = 'opacity 1s ease 0.5s'; setTimeout(() => area.style.opacity = '1', 50); }
    dots.forEach(d => { d.classList.remove('animate'); void d.offsetWidth; d.classList.add('animate'); });
    bars.forEach(b => { b.style.width = '0'; });
    requestAnimationFrame(() => {
      setTimeout(() => {
        bars.forEach(b => { b.style.width = b.dataset.width; });
      }, 200);
    });
  }
